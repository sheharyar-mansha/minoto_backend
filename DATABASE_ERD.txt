--------------------------------------------------------------------------------
ENTITY LIST (TABLES)
--------------------------------------------------------------------------------

  users                  PK: id (varchar 36)
  members                PK: id  |  FK: user_id → users.id (CASCADE)
  meetings               PK: id  |  FK: user_id → users.id (CASCADE)
  meeting_members        PK: (meeting_id, member_id)  |  FKs → meetings, members (CASCADE)
  meeting_live_sessions  PK: user_id  |  FK: user_id → users.id, meeting_id → meetings.id (CASCADE)


--------------------------------------------------------------------------------
RELATIONSHIPS (CARDINALITY)
--------------------------------------------------------------------------------

  User 1 ──< Member              (one account owns many directory members)
  User 1 ──< Meeting             (one account owns many meetings)
  User 1 ──o| MeetingLiveSession (at most one live-session row per user; PK = user_id)

  Meeting N ──< meeting_members >── M Member   (many-to-many via junction table)

  MeetingLiveSession >── Meeting   (each session row points to one meeting)


--------------------------------------------------------------------------------
COLUMNS (QUICK REFERENCE)
--------------------------------------------------------------------------------

  users
    id, email (UNIQUE), full_name, hashed_password,
    default_export_format, include_timestamps_default,
    avatar_url (nullable; added migration 002),
    created_at, updated_at

  members
    id, user_id (FK users), name, email, role, member_since, saved_to_directory,
    avatar_url, has_voice_sample, voice_file_path, voice_duration_seconds,
    created_at, updated_at

  meetings
    id, user_id (FK users), title, meeting_date, start_time, duration_minutes,
    status, conducted_at, created_at, updated_at

  meeting_members  (ORM: MeetingMemberLink)
    meeting_id (FK meetings, part of PK),
    member_id  (FK members, part of PK),
    is_new_for_meeting

  meeting_live_sessions
    user_id (PK, FK users),
    meeting_id (FK meetings, indexed),
    title, elapsed_seconds, is_paused, running_since_ms, total_target_seconds
